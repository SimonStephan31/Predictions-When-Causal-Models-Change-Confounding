Function computing P(e|c) under the preventive CCCF CBN

p_ec_ccm_cf <- function(w_ce, w_ae, w_ac, b_a, check = TRUE) {

  if (check) {
    if (any(is.na(c(w_ce, w_ae, w_ac, b_a))))
      stop("Inputs must not be NA.")
    
    if (any(c(w_ce, w_ae, w_ac, b_a) < 0 | 
            c(w_ce, w_ae, w_ac, b_a) > 1)) {
      warning("Inputs are typically probabilities/weights in [0,1]. Proceeding anyway.")
    }

    denom <- 1 - b_a * w_ac
    if (isTRUE(all.equal(denom, 0)))
      stop("Invalid inputs: denominator equals 0 (division by zero).")
  }

  frac <- (b_a * (1 - w_ac)) / (1 - b_a * w_ac)

  w_ce + (w_ae - w_ce) * frac
}

Function computing P(e|c) under the generative CCCF CBN

p_ec_cccf <- function(w_ce, w_ae, w_ac, b_a, b_c, check = TRUE) {
  if (check) {
    if (any(is.na(c(w_ce, w_ae, w_ac, b_a, b_c))))
      stop("Inputs must not be NA.")
    
    if (any(c(w_ce, w_ae, w_ac, b_a, b_c) < 0 | 
            c(w_ce, w_ae, w_ac, b_a, b_c) > 1)) {
      warning("Inputs are typically probabilities/weights in [0, 1]. Proceeding anyway.")
    }

    denom <- w_ac*b_a + b_c - w_ac*b_c*b_a
    if (isTRUE(all.equal(denom, 0)))
      stop("Invalid inputs: denominator equals 0 (division by zero).")
  }

  numerator   <- w_ac*b_a + b_c*b_a - w_ac*b_c*b_a
  denominator <- w_ac*b_a + b_c - w_ac*b_c*b_a

  w_ce + (1 - w_ce) * w_ae * (numerator / denominator)
}

# --- 2. Define the Margin of Error ---
margin_of_error <- 3 # 3% margin (scale from 0 to 100)

# --- 3. Cluster Participants ---
tdata_clusters <- tdata %>%
  mutate(
    # Calculate the difference for easier understanding
    rating_difference = case_09 - case_01,
    
    cluster = case_when(
      # Condition 1: No change (absolute difference is within the margin)
      # This condition must come FIRST to give precedence to the margin
      abs(rating_difference) <= margin_of_error ~ "no change",
      
      # Condition 2: Increase 
      rating_difference > margin_of_error ~ "increase",
      
      # Condition 3: Decrease
      rating_difference < -margin_of_error ~ "decrease",
      
      # Optional: A catch-all for any cases not covered (shouldn't happen with these conditions)
      TRUE ~ NA_character_ 
    )
  )
