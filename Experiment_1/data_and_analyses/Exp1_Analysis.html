library(dplyr)

# --- 2. Define the Margin of Error ---
margin_of_error <- 3 # 3% margin (scale from 0 to 100)

# --- 3. Cluster Participants ---
tdata_clusters <- tdata %>%
  mutate(
    # Calculate the difference for easier understanding
    rating_difference = case_2 - case_1,
    
    cluster = case_when(
      # Condition 1: No change (absolute difference is within the margin)
      # This condition must come FIRST to give precedence to the margin
      abs(rating_difference) <= margin_of_error ~ "no change",
      
      # Condition 2: Increase (cf_rating is sufficiently greater than ic_rating)
      # This means cf_rating - ic_rating > margin_of_error
      rating_difference > margin_of_error ~ "increase",
      
      # Condition 3: Decrease (ic_rating is sufficiently greater than cf_rating)
      # This means ic_rating - cf_rating > margin_of_error, or rating_difference < -margin_of_error
      rating_difference < -margin_of_error ~ "decrease",
      
      # Optional: A catch-all for any cases not covered (shouldn't happen with these conditions)
      TRUE ~ NA_character_ 
    )
  )

cluster_summary <- tdata_clusters %>%
  count(comparison, cluster, name = "n") %>%
  group_by(comparison) %>%
  mutate(
    n_total = sum(n),
    p = n / n_total,
    percentage = 100 * p
  ) %>%
  ungroup() %>%
  # Wilson 95% CI for each bar
  bind_cols(
    binom.confint(
      x = .$n,
      n = .$n_total,
      methods = "wilson"
    ) %>%
      transmute(
        ci_low  = 100 * lower,
        ci_high = 100 * upper
      )
  )
